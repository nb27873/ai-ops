mesma feed) a contagem irá falhar sempre.

Este que agora está a Deleted (ficheiro gerado na exec do DMaap)
carregou 2 ficheiros
(COPS_RETENTION_PL_DETAIL_LOAD_20260509000000_MSISDN.csv e
COPS_RETENTION_PL_DETAIL_LOAD_20260510000000_MSISDN.csv)

 

Logo o Numero de Linhas é a soma de ambos os ficheiros (vamos assumir
que tem 100 linhas cada 1) aqui a contagem seria de 200 Nr_lines, mas
quando faz o validate row count é por ficheiro, logo iria dar 200 vs 100
que dará mismatch no numero de linhas.\
 

Para resolver isto, temos de processar outra vez:

**ROW_VALIDATION: (por dia)**

 

WITH expected_files AS (

SELECT

REPLACE(file_name, \'\*\', \'%20260509%\') AS expected_file_pattern

FROM
vf-pt-datahub-prd-mgmt.vfpt_dh_mgmt_dmaap_metadata_s.bde_rd_da_cops_feeds

WHERE ds_area = \'RETENTION\'

),

log_counts AS (

SELECT

file_name AS filename,

SUM(row_count) AS log_row_count

FROM \`vf-pt-datahub.vfpt_dh_lake_cops_others_s.cops_logs_files\`

WHERE EXISTS (

SELECT 1 FROM expected_files ef WHERE file_name LIKE
ef.expected_file_pattern

)

AND status = \'SUCCESS\'

GROUP BY file_name

),

file_counts AS (

SELECT

tf_original_file_name AS filename,

SUM(nr_lines_append) AS file_nr_lines_sent,

SUM(nr_lines_original_file) AS file_nr_lines_original,

SUM(record_count_gcp) AS file_nr_lines_bq

FROM
\`vf-pt-datahub.vfpt_dh_lake_cops_others_s.audit_files_dmaap_processing_v\`

WHERE EXISTS (

SELECT 1 FROM expected_files ef WHERE tf_original_file_name LIKE
ef.expected_file_pattern

)

GROUP BY tf_original_file_name

),

join_counts AS (

SELECT

l.filename,

l.log_row_count,

f.file_nr_lines_sent,

f.file_nr_lines_original,

f.file_nr_lines_bq,

CASE

WHEN (l.log_row_count != f.file_nr_lines_sent

OR l.log_row_count != f.file_nr_lines_original

OR l.log_row_count != f.file_nr_lines_bq)

THEN \'NOK\'

ELSE \'OK\'

END AS file_status

FROM log_counts l

INNER JOIN file_counts f

ON l.filename = f.filename

),

final_status AS (

SELECT

CASE

WHEN COUNT(CASE WHEN file_status = \'NOK\' THEN 1 END) \> 0

THEN \'NOK: \' \|\| STRING_AGG(CASE WHEN file_status = \'NOK\' THEN
filename END, \', \')

ELSE \'OK\'

END AS status

FROM join_counts

)

SELECT status FROM final_status;
